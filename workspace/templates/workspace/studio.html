from lyzr import AgentWorkflow, RouterNode, AgentNode

workflow = AgentWorkflow(id="wf_8f29d10a")

classifier = RouterNode(
    name="IntentClassifier",
    prompt="Classify whether message is flight_info or itinerary",
    output_schema={"classification": ["flight_info", "itinerary"]}
)

flight_agent = AgentNode(
    name="FlightAgent",
    tools=["web_search"],
    widget_output="FlightWidgetCard"
)

workflow.connect(start_node="Start", target=classifier)
workflow.branch(classifier, condition="flight_info", target=flight_agent)

# Deploy directly using Workflow ID
response = workflow.run(input_prompt="Find flights from SFO to Tokyo")
print(response.widget_data)
